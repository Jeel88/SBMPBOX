Q: Create Student_Report table to store results?

SQL
CREATE TABLE Student_Report (
  Student_ID INT,
  Student_Name VARCHAR(50),
  Marks DECIMAL(5,2),
  Result VARCHAR(10)
);
Q: Create stored procedure using CURSOR to calculate PASS/FAIL (cutoff 40)?

SQL
DELIMITER //
CREATE PROCEDURE GenerateStudentReport()
BEGIN
  DECLARE v_ID INT;
  DECLARE v_Name VARCHAR(50);
  DECLARE v_Marks DECIMAL(5,2);
  DECLARE v_Result VARCHAR(10);
  DECLARE done INT DEFAULT 0;

  DECLARE student_cursor CURSOR FOR
    SELECT Student_ID, Student_Name, Marks FROM Student ORDER BY Student_ID;

  DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

  DELETE FROM Student_Report;

  OPEN student_cursor;

  read_loop: LOOP
    FETCH student_cursor INTO v_ID, v_Name, v_Marks;
    IF done = 1 THEN
      LEAVE read_loop;
    END IF;

    IF v_Marks >= 40 THEN
      SET v_Result = 'PASS';
    ELSE
      SET v_Result = 'FAIL';
    END IF;

    INSERT INTO Student_Report VALUES (v_ID, v_Name, v_Marks, v_Result);
  END LOOP;

  CLOSE student_cursor;
END //
DELIMITER ;
Q: Run procedure, view report & get summary count by result?

SQL
CALL GenerateStudentReport();

-- Display report
SELECT * FROM Student_Report;

-- Summary count
SELECT Result, COUNT(*) AS Number_Of_Students
FROM Student_Report
GROUP BY Result;
