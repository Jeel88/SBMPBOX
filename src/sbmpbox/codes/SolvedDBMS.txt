BEFORE UPDATE ON Employee
FOR EACH ROW
BEGIN
    IF OLD.salary <> NEW.salary THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Error: Updating the Salary attribute is not allowed!';

BEFORE INSERT ON Employee
FOR EACH ROW
BEGIN
    -- DAYOFWEEK: 1 = Sunday, 7 = Saturday
    IF DAYOFWEEK(CURDATE()) IN (1, 7) THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Error: Inserting records into Employee on weekends is restricted!';
    END IF;
END //

-- Q14] Stored Procedure: Update_Account_Balance (MySQL Equivalent)
-- =========================================================================
DROP TABLE IF EXISTS Account;
CREATE TABLE Account (
    Acc_No INT PRIMARY KEY,
    Cust_Name VARCHAR(30),
    Balance DECIMAL(10,2),
    Acc_Type VARCHAR(15)
);

INSERT INTO Account VALUES
(1001, 'Amit Sharma', 10000.00, 'Savings'),
(1002, 'Neha Verma', 1500.00, 'Savings');

DELIMITER //

CREATE PROCEDURE Update_Account_Balance(
    IN p_acc_no INT,
    IN p_trans_type CHAR(1),
    IN p_amount DECIMAL(10,2)
)
BEGIN
    DECLARE v_balance DECIMAL(10,2);

    SELECT Balance INTO v_balance FROM Account WHERE Acc_No = p_acc_no;

    IF p_trans_type = 'D' THEN
        UPDATE Account SET Balance = Balance + p_amount WHERE Acc_No = p_acc_no;
        SELECT CONCAT('Deposited ', p_amount, '. New Balance: ', v_balance + p_amount) AS Status;
    ELSEIF p_trans_type = 'W' THEN
        IF v_balance >= p_amount THEN
            UPDATE Account SET Balance = Balance - p_amount WHERE Acc_No = p_acc_no;
            SELECT CONCAT('Withdrawn ', p_amount, '. New Balance: ', v_balance - p_amount) AS Status;
        ELSE
            SELECT 'Insufficient Balance' AS Status;
        END IF;
    END IF;
END //

DELIMITER ;

-- Execute procedure calls
CALL Update_Account_Balance(1001, 'D', 5000);
CALL Update_Account_Balance(1002, 'W', 2000); -- Displays Insufficient Balance

(1, 'Rohan', 80, 75, 85),
(2, 'Sneha', 90, 88, 92),
(3, 'Vikas', 60, 65, 70),
(4, 'Ananya', 78, 82, 80),

DELIMITER //

CREATE PROCEDURE Calc_Student_Result(IN p_roll INT, IN p_max_marks INT)
BEGIN
    DECLARE v_s1, v_s2, v_s3, v_total INT;
    DECLARE v_pct DECIMAL(5,2);
    DECLARE v_not_found INT DEFAULT 0;

    -- Handler for NO_DATA_FOUND
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET v_not_found = 1;

    SELECT Sub1, Sub2, Sub3 INTO v_s1, v_s2, v_s3 
    FROM Student_Marks WHERE Roll_No = p_roll;

    IF v_not_found = 1 THEN
        SELECT 'Student record does not exist' AS Error_Message;
    ELSEIF p_max_marks = 0 THEN
        -- Handling ZERO_DIVIDE explicitly
        SELECT 'Error: Division by zero in percentage calculation' AS Error_Message;
    ELSE
        SET v_total = v_s1 + v_s2 + v_s3;
        SET v_pct = (v_total / p_max_marks) * 100;
        UPDATE Student_Marks SET Total = v_total, Percentage = v_pct WHERE Roll_No = p_roll;
        SELECT * FROM Student_Marks WHERE Roll_No = p_roll;
    END IF;
END //

DELIMITER ;

CALL Calc_Student_Result(1, 300);   -- Existing student
CALL Calc_Student_Result(99, 300);  -- Non-existing student

DELIMITER //

CREATE PROCEDURE Get_City_Orders(IN p_city VARCHAR(20))
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE v_cname VARCHAR(30);
    DECLARE v_oid INT;
    DECLARE v_amt DECIMAL(10,2);
    DECLARE v_row_count INT DEFAULT 0;

    DECLARE cur_orders CURSOR FOR 
        SELECT c.Cust_Name, o.Order_ID, o.Order_Amount
        FROM Customer c
        JOIN Orders o ON c.Cust_ID = o.Cust_ID
        WHERE c.City = p_city;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

    DROP TEMPORARY TABLE IF EXISTS Temp_City_Report;
    CREATE TEMPORARY TABLE Temp_City_Report (Customer_Name VARCHAR(30), Order_ID INT, Order_Amount DECIMAL(10,2));

    OPEN cur_orders;
    read_loop: LOOP
        FETCH cur_orders INTO v_cname, v_oid, v_amt;
        IF done = 1 THEN
            LEAVE read_loop;
        END IF;
        INSERT INTO Temp_City_Report VALUES (v_cname, v_oid, v_amt);
        SET v_row_count = v_row_count + 1;
    END LOOP;
    CLOSE cur_orders;

    SELECT * FROM Temp_City_Report;
    SELECT v_row_count AS Total_Matching_Orders;
END //

DELIMITER ;

-- Q23] Complex Joins with Multi-Table Schema (Sailors, Boats, Reserves)
-- =========================================================================
DROP TABLE IF EXISTS Reserves;
DROP TABLE IF EXISTS Sailors;
DROP TABLE IF EXISTS Boats;

CREATE TABLE Sailors (
    sid INT PRIMARY KEY,
    sname VARCHAR(20),
    rating INT,
    age DECIMAL(4,1)
);

CREATE TABLE Boats (
    bid INT PRIMARY KEY,
    bname VARCHAR(20),
    color VARCHAR(15)
);

CREATE TABLE Reserves (
    sid INT,
    bid INT,
    day DATE,
    PRIMARY KEY (sid, bid, day),
    FOREIGN KEY (sid) REFERENCES Sailors(sid),
    FOREIGN KEY (bid) REFERENCES Boats(bid)
);

INSERT INTO Sailors VALUES (22, 'Dustin', 7, 45.0), (29, 'Brutus', 1, 33.0), (31, 'Lubber', 8, 55.5), (32, 'Andy', 8, 25.5);
INSERT INTO Boats VALUES (101, 'Interlake', 'Blue'), (102, 'Interlake', 'Red'), (103, 'Clipper', 'Green'), (104, 'Marine', 'Red');
INSERT INTO Reserves VALUES 
(22, 102, '2026-10-10'),
(22, 104, '2026-10-10'), -- Reserved 2 distinct red boats on same day
(31, 102, '2026-11-10'),
(31, 103, '2026-11-12'); -- Reserved both Red and Green

-- 1. Sailors who reserved a 'Red' boat but NOT a 'Green' boat
SELECT DISTINCT s.sname
FROM Sailors s
JOIN Reserves r ON s.sid = r.sid
JOIN Boats b ON r.bid = b.bid
WHERE b.color = 'Red'
  AND s.sid NOT IN (
      SELECT r2.sid FROM Reserves r2 JOIN Boats b2 ON r2.bid = b2.bid WHERE b2.color = 'Green'
  );

-- 2. Sailors who reserved at least two distinct boats on the same day
SELECT s.sname
FROM Sailors s
JOIN Reserves r ON s.sid = r.sid
GROUP BY s.sid, s.sname, r.day
HAVING COUNT(DISTINCT r.bid) >= 2;

-- 3. FULL OUTER JOIN between Sailors and Reserves (Simulated via UNION in MySQL)
SELECT s.sid, s.sname, r.bid, r.day
FROM Sailors s
LEFT JOIN Reserves r ON s.sid = r.sid
UNION
SELECT s.sid, s.sname, r.bid, r.day
FROM Sailors s
RIGHT JOIN Reserves r ON s.sid = r.sid;

    Item_ID INT PRIMARY KEY,
    Item_Name VARCHAR(30),
    Quantity INT
);

START TRANSACTION;
INSERT INTO Inventory VALUES (1, 'Keyboard', 50), (2, 'Mouse', 80);
COMMIT;

-- Start new transaction with Savepoints
START TRANSACTION;
INSERT INTO Inventory VALUES (3, 'Monitor', 20);
SAVEPOINT SP1;

UPDATE Inventory SET Quantity = 45 WHERE Item_ID = 1;
SAVEPOINT SP2;

DELETE FROM Inventory WHERE Item_ID = 2;

ROLLBACK TO SP1;

SELECT * FROM Inventory;
COMMIT;
