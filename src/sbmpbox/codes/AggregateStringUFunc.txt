Q: Create DB, Student table & insert sample records?

SQL
CREATE DATABASE CollegeDB;
USE CollegeDB;

CREATE TABLE Student (
  Student_ID INT PRIMARY KEY,
  Student_Name VARCHAR(50) NOT NULL,
  Department VARCHAR(30),
  Marks DECIMAL(5,2)
);

Q: Display students with uppercase names ordered by marks DESC?

SQL
SELECT
  Student_ID,
  UPPER(Student_Name) AS Student_Name,
  Department,
  Marks
FROM Student
ORDER BY Marks DESC;
Q: Summary report using aggregate functions (COUNT, AVG, MAX, MIN)?

SQL
SELECT
  COUNT(*) AS Total_Students,
  ROUND(AVG(Marks), 2) AS Average_Marks,
  MAX(Marks) AS Highest_Marks,
  MIN(Marks) AS Lowest_Marks
FROM Student;
Q: Department-wise summary using GROUP BY?

SQL
SELECT
  Department,
  COUNT(*) AS Total_Students,
  ROUND(AVG(Marks), 2) AS Average_Marks,
  MAX(Marks) AS Highest_Marks
FROM Student
GROUP BY Department;
Q: Create UDF to return PASS/FAIL (cutoff 40)?

SQL
DELIMITER //
CREATE FUNCTION CalculateResult(p_marks DECIMAL(5,2))
RETURNS VARCHAR(10)
DETERMINISTIC
NO SQL
BEGIN
  IF p_marks >= 40 THEN
    RETURN 'PASS';
  ELSE
    RETURN 'FAIL';
  END IF;
END //
DELIMITER ;
Q: Display final report using the UDF sorted by Department, Name?

SQL
SELECT
  Student_ID,
  Student_Name,
  Department,
  Marks,
  CalculateResult(Marks) AS Result
FROM Student
ORDER BY Department, Student_Name;
Q: Display pass/fail student count using the UDF with GROUP BY?

SQL
SELECT
  CalculateResult(Marks) AS Result,
  COUNT(*) AS Number_Of_Students
FROM Student
GROUP BY CalculateResult(Marks);
