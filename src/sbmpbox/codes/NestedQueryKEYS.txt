Q: Create DB & tables with Primary/Foreign keys?

SQL
CREATE DATABASE CollegeDB;
USE CollegeDB;

CREATE TABLE Department (
  DeptID INT PRIMARY KEY,
  DeptName VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE Student (
  StudentID INT PRIMARY KEY,
  StudentName VARCHAR(50) NOT NULL,
  DeptID INT,
  Marks DECIMAL(5,2),
  FOREIGN KEY (DeptID) REFERENCES Department(DeptID)
);

Q: Find students scoring above average marks?

SQL
SELECT * FROM Student
WHERE Marks > (SELECT AVG(Marks) FROM Student);
Q: Find students in 'IT' department using subquery?

SQL
SELECT * FROM Student
WHERE DeptID = (SELECT DeptID FROM Department WHERE DeptName = 'IT');
Q: Find student(s) with the highest marks?

SQL
SELECT * FROM Student
WHERE Marks = (SELECT MAX(Marks) FROM Student);
Q: Students scoring more than ANY student in IT?

SQL
SELECT * FROM Student
WHERE Marks > ANY (
  SELECT S.Marks FROM Student S
  JOIN Department D ON S.DeptID = D.DeptID
  WHERE D.DeptName = 'IT'
);
Q: Students scoring more than ALL students in IT?

SQL
SELECT * FROM Student
WHERE Marks > ALL (
  SELECT S.Marks FROM Student S
  JOIN Department D ON S.DeptID = D.DeptID
  WHERE D.DeptName = 'IT'
);
Q: Students whose department exists using EXISTS?

SQL
SELECT * FROM Student S
WHERE EXISTS (
  SELECT 1 FROM Department D
  WHERE D.DeptID = S.DeptID
);
